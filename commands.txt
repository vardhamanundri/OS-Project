./campuscompute sched compare

# S14 - Banker's safe stat

# S16 - normal worker process
./campuscompute worker normal

# S17 - simulated worker failure
./campuscompute worker all

# S18 - resource limits
./campuscompute rlimit all

./campuscompute cleanup; ls /dev/shm | grep -i campus; pgrep -l campuscompute; echo "no leftovers"
